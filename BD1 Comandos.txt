CREATE TABLE CARGO (codigo int, nome varchar(20) not null, primary key(codigo))

CREATE TABLE EMPREGADO (codigo int not null identity, nome varchar(45) not null, sexo char(1) default 'F',
CPF int unique not null, codcargo int, telefone varchar(10),
dataatualizacao smalldatetime default getdate(), primary key (codigo),
foreign key (codcargo) references CARGO(codigo) on update cascade, check (sexo in ('F', 'M')))

SELECT nome, salario AS salario_mensal, codcargo as cargo FROM EMPREGADO

SELECT * FROM EMPREGADO WHERE telefone is null

SELECT nome, salario AS salario_mensal, salario * 12 AS salario_anual FROM EMPREGADO

SELECT EMPREGADO.nome as nome,
EMPREGADO.codcargo as cargo,
CARGO.nome as nome_cargo
FROM EMPREGADO INNER JOIN CARGO
ON EMPREGADO.codcargo = CARGO.codigo

SELECT EMPREGADO.nome as nome,
CARGO.nome as nome_cargo,
funcao.nome as nome_funcao
FROM EMPREGADO INNER JOIN CARGO
ON EMPREGADO.codcargo = CARGO.codigo
INNER JOIN Funcao
ON EMPREGADO.codfuncao = Funcao.codigo

SELECT e.nome as nome, e.codcargo as cargo,
c.nome as nome_cargo
FROM EMPREGADO e LEFT OUTER JOIN CARGO c
ON e.codcargo = c.codigo

SELECT e.nome as nome,
c.nome as nome_cargo,
f.nome as nome_funcao
FROM EMPREGADO e INNER JOIN CARGO c
ON e.codcargo = c.codigo
INNER JOIN Funcao f
ON e.codfuncao = f.codigo

SELECT e.nome as nome, e.codcargo as cargo,
c.nome as nome_cargo
FROM CARGO c LEFT OUTER JOIN EMPREGADO e
ON e.codcargo = c.codigo

SELECT e.nome as nome, e.codcargo as codigo,
c.nome as nome_cargo
FROM EMPREGADO e RIGHT OUTER JOIN CARGO c
ON e.codcargo = c.codigo

SELECT e.nome as nome, e.codcargo as codigo,
c.nome as nome_cargo FROM EMPREGADO e
FULL OUTER JOIN CARGO c ON e.codcargo = c.codigo

SELECT sexo, AVG(salario) as media_salarial
FROM EMPREGADO
GROUP BY sexo

SELECT codcargo, sexo, AVG(salario)
FROM EMPREGADO
GROUP BY codcargo, sexo

SELECT codcargo, sexo, AVG(salario)
FROM EMPREGADO
GROUP BY codcargo, sexo
HAVING codcargo > 2

SELECT codcargo, sexo, AVG(salario) as media_salarial
FROM EMPREGADO
WHERE salario > 8000.00
GROUP BY codcargo, sexo
HAVING codcargo > 2
