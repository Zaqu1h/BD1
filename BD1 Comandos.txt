CREATE TABLE CARGO (codigo int, nome varchar(20) not null, primary key(codigo))

CREATE TABLE EMPREGADO (codigo int not null identity, nome varchar(45) not null, sexo char(1) default 'F',
CPF int unique not null, codcargo int, telefone varchar(10),
dataatualizacao smalldatetime default getdate(), primary key (codigo),
foreign key (codcargo) references CARGO(codigo) on update cascade, check (sexo in ('F', 'M')))
